  2. In `src/battle/StatusEngine.ts`, add an `inflictStatus()` function after the existing `inflictSleep()`:

  ```typescript
  export function inflictStatus(
    pokemon: ActivePokemon,
    status: string,
    actor: "player" | "opponent",
  ): { newPokemon: ActivePokemon; log: TurnLogEntry | null } {
    // Cannot inflict if already statused
    if (pokemon.status !== "") {
      return { newPokemon: pokemon, log: null };
    }
    // Cannot inflict on fainted Pokemon
    if (pokemon.fainted) {
      return { newPokemon: pokemon, log: null };
    }

    let newPokemon: ActivePokemon;

    switch (status) {
      case "slp":
        newPokemon = inflictSleep(pokemon);  // reuse existing — rolls 2-4 turns
        break;
      case "tox":
        newPokemon = { ...pokemon, status: "tox" as const, toxicCounter: 1 };
        break;
      case "brn":
      case "par":
      case "psn":
      case "frz":
        newPokemon = { ...pokemon, status: status as StatusCondition };
        break;
      default:
        return { newPokemon: pokemon, log: null }; // unknown status, no-op
    }

    const log: TurnLogEntry = {
      kind: "STATUS_INFLICT",
      actor,
      detail: { status, pokemon: pokemon.name },
    };

    return { newPokemon, log };
  }
  ```

  Import `StatusCondition` type from `../types/battleState.js` (it is already in scope if the file already imports `ActivePokemon` and `TurnLogEntry`).

  Key rules:
  - If pokemon.status is non-empty, return unchanged (no double-status in Gen 9).
  - For "tox", set toxicCounter to 1 (Pitfall 7 from decisions).
  - For "slp", delegate to existing inflictSleep() which rolls 2-4 turns.
  - Return a TurnLogEntry with kind "STATUS_INFLICT" so UI can display it.
