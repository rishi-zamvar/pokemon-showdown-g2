Add these methods to the exported DataLoader object:

```typescript
// Load match-scoped data for exactly the 12 battle Pokemon
async loadMatchData(pokemonIds: string[]): Promise<void> {
  // Build match Pokemon map from global pokedex (no network call — already in memory)
  _matchPokemon = new Map(
    pokemonIds
      .map(id => [id, _pokedex.get(id)] as const)
      .filter((entry): entry is [string, PokemonEntry] => entry[1] !== undefined)
  );

  // Lazy-load learnsets from public/ — only extract the 12 match Pokemon
  const raw: Record<string, Record<string, unknown>> = await fetch('/data/learnsets.json').then(r => r.json());
  _matchLearnsets = new Map(
    pokemonIds
      .filter(id => id in raw)
      .map(id => [id, raw[id]])
  );
  // raw is now eligible for GC — only the 12 Pokemon's learnsets are retained
},

// Clear all match data — call when battle ends
clearMatchData(): void {
  _matchPokemon = null;
  _matchLearnsets = null;
},

// Match-scoped accessors
getMatchPokemon(id: string): PokemonEntry | undefined {
  return _matchPokemon?.get(id);
},

getMatchLearnset(id: string): Record<string, unknown> | undefined {
  return _matchLearnsets?.get(id);
},

isMatchLoaded(): boolean {
  return _matchPokemon !== null;
},
```

**Important considerations:**
- `loadMatchData` accepts lowercase Pokemon IDs (same format as pokedex keys)
- The `raw` variable from fetch is intentionally NOT stored — only the 12 match entries are kept
- `clearMatchData` sets both maps to null so the GC can reclaim memory
- If learnsets fetch fails, let the error propagate — caller handles UI feedback
