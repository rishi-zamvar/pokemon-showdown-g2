2. Add a module-level helper to shift sprite content within a BMP:
```typescript
/**
 * Create a shifted version of a sprite BMP for slide animation.
 * Reads the pixel data from the original BMP, shifts all pixels
 * by `offsetX` pixels to the right within the 96x96 canvas,
 * and re-encodes as a new BMP.
 *
 * This works because the G2 SDK accepts full BMP images — we shift
 * the sprite content within the image rather than moving the container.
 */
function shiftSpriteBmp(original: Uint8Array, offsetX: number, size: number): Uint8Array {
  // BMP header is 62 bytes, pixel data follows in bottom-up row order
  // Each row is ceil(size/8) bytes padded to 4-byte boundary
  const BMP_HEADER = 62;
  const rowStride = Math.ceil(Math.ceil(size / 8) / 4) * 4;

  // Decode 1-bit pixels from original BMP
  const pixels = new Uint8Array(size * size);
  for (let y = 0; y < size; y++) {
    const srcRow = size - 1 - y; // BMP is bottom-up
    const rowOffset = BMP_HEADER + y * rowStride;
    for (let x = 0; x < size; x++) {
      const byteIdx = rowOffset + Math.floor(x / 8);
      const bitIdx = 7 - (x % 8);
      if (original[byteIdx]! & (1 << bitIdx)) {
        pixels[srcRow * size + x] = 1;
      }
    }
  }

  // Shift pixels right by offsetX (pixels that fall off the right edge are clipped)
  const shifted = new Uint8Array(size * size);
  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const srcX = x - offsetX;
      if (srcX >= 0 && srcX < size) {
        shifted[y * size + x] = pixels[y * size + srcX]!;
      }
    }
  }

  return encodeBmp(shifted, size, size);
}
```

```typescript
private async playEntry(
  entry: TurnLogEntry,
  prevState: BattleState,
  nextState: BattleState,
  glass: GlassAdapter,
): Promise<void> {
  const fullText = formatEntry(entry, prevState);

  // Shared mutable state for concurrent text + HP updates
  let currentText = '';
  let hpStatus = '';

  const updateDisplay = async (): Promise<void> => {
    await glass.updateText(currentText, { left: hpStatus, right: '' });
  };

  // SWITCH entries: run animation FIRST, then typewriter text SEQUENTIALLY
  // (they both use glass updates and would race if concurrent)
  if (entry.kind === 'SWITCH') {
    const switchedSide: Side = entry.actor;
    // Step 1: Play 3-frame switch animation + sprite slide
    await this.switchAnimation(switchedSide, nextState, glass);
    // Step 2: Typewriter the "Go! Pokemon!" / "Foe sent out Pokemon!" text
    await this.typewriterText(fullText, glass, (text) => {
      currentText = text;
    }, () => updateDisplay());
    return;
  }

  // All other entries: concurrent tasks via Promise.all (existing pattern)
  const tasks: Promise<void>[] = [];

  // Typewriter text -- always runs
  tasks.push(this.typewriterText(fullText, glass, (text) => {
    currentText = text;
  }, () => updateDisplay()));

  if (entry.kind === 'DAMAGE') {
    // Damage: target is the opposite side of the actor
    const targetSide: Side = entry.actor === 'player' ? 'opponent' : 'player';

    tasks.push(this.spriteFlash(targetSide, 'hit', prevState, glass));
    tasks.push(this.hpDrain(
      targetSide, entry, prevState, glass,
      (status) => { hpStatus = status; },
      () => updateDisplay(),
    ));
  } else if (entry.kind === 'MISS') {
    // Miss: attacker flashes (disappear/reappear)
    tasks.push(this.spriteFlash(entry.actor, 'miss', prevState, glass));
  } else if (entry.kind === 'FAINT') {
    // Faint: determine which side fainted by matching pokemon name
    const faintedName = entry.detail.pokemon as string;
    const faintedSide = this.findFaintedSide(faintedName, prevState);
    if (faintedSide) {
      tasks.push(this.spriteFlash(faintedSide, 'faint', prevState, glass));
    }
  } else if (entry.kind === 'HEAL') {
    // Heal: HP bar fills up
    const healSide: Side = entry.actor;
    tasks.push(this.hpHeal(
      healSide, prevState, nextState, glass,
      (status) => { hpStatus = status; },
      () => updateDisplay(),
    ));
  } else if (entry.kind === 'STATUS_TICK') {
    const tickSide: Side = entry.actor;
    const tickDamage = entry.detail.targetHp as number | undefined;
    if (tickDamage !== undefined && tickDamage >= 0) {
      tasks.push(this.spriteFlash(tickSide, 'hit', prevState, glass));
    }
  }

  await Promise.all(tasks);
}
```

4. Add the `switchAnimation` private method with REAL sprite slide:

```typescript
/**
 * Play 3-frame switch-in animation: pokeball -> filled square -> checkbox -> sprite slide -> real sprite.
 * The "slide" is implemented by generating shifted BMP versions of the real sprite
 * (offset 2-3px within the 96x96 canvas) and displaying them in sequence.
 */
private async switchAnimation(
  side: Side,
  nextState: BattleState,
  glass: GlassAdapter,
): Promise<void> {
  const containerID = side === 'opponent'
    ? CONTAINER_IDS.oppSprite
    : CONTAINER_IDS.playerSprite;
  const containerName = side === 'opponent'
    ? CONTAINER_NAMES.oppSprite
    : CONTAINER_NAMES.playerSprite;

  const updateImg = async (data: Uint8Array): Promise<void> => {
    await glass.updateImage(containerID, containerName, data);
  };

  if (this.cancelled) return;

  // Frame 1: Pokeball
  await updateImg(getPokeballFrame());
  if (this.cancelled) return;

  // Frame 2: Filled square (energy burst)
  await updateImg(getFilledSquareFrame());
  if (this.cancelled) return;

  // Frame 3: Checkbox (materialization)
  await updateImg(getCheckboxFrame());
  if (this.cancelled) return;

  // Frame 4+: Sprite slide — show shifted versions of the real sprite
  const newSprite = getSpriteForSide(nextState, side);
  if (newSprite) {
    const spriteSize = SPRITE_CONFIG.SIZE; // 96
    const slidePx = ANIM.SWITCH_SLIDE_PX; // 3
    const slideFrames = ANIM.SWITCH_SLIDE_FRAMES; // 2

    // Generate slide frames: sprite shifted left by decreasing amounts
    // Frame 1: shifted left by slidePx*slideFrames (e.g. 6px)
    // Frame 2: shifted left by slidePx (e.g. 3px)
    // Final: real sprite at correct position (0px offset)
    for (let f = slideFrames; f >= 1; f--) {
      if (this.cancelled) break;
      const offset = -(slidePx * f); // negative = shifted left (slides in from left)
      const shiftedBmp = shiftSpriteBmp(newSprite, offset, spriteSize);
      await updateImg(shiftedBmp);
      await this.delay(ANIM.FLASH_FRAME_MS); // reuse flash frame timing
    }

    // Final frame: real sprite at correct position
    if (!this.cancelled) {
      await updateImg(newSprite);
    }
  }
}
```

```typescript
/**
 * Reverse HP drain — HP bar fills UP progressively.
 * Used for HEAL entries (Leftovers, Wish, Recover, etc.)
 *
 * HP calculation: Read startHp from prevState.currentHp (same side as the heal).
 * Read endHp from nextState.currentHp (after the heal is applied).
 * This avoids the percentage-based log string problem — both player and opponent
 * use actual HP values derived from the sim.
 */
private async hpHeal(
  side: Side,
  prevState: BattleState,
  nextState: BattleState,
  glass: GlassAdapter,
  setHpStatus: (status: string) => void,
  triggerUpdate: () => Promise<void>,
): Promise<void> {
  const prevMon = getActiveMon(prevState, side);
  const nextMon = getActiveMon(nextState, side);

  const startHp = prevMon.currentHp;
  const endHp = nextMon.currentHp;
  const healAmount = endHp - startHp;

  if (healAmount <= 0) return; // No actual heal (or damage masquerading as heal)

  const steps = Math.min(healAmount, ANIM.MAX_HP_STEPS);
  if (steps <= 0) return;

  for (let i = 0; i <= steps; i++) {
    if (this.cancelled) break;

    const currentHp = Math.round(startHp + (healAmount * i / steps));
    const tempMon: ActivePokemon = { ...prevMon, currentHp, fainted: false };

    setHpStatus(formatHpBar(tempMon));
    await triggerUpdate();
    await this.delay(ANIM.HP_FRAME_MS);
  }
}
```
