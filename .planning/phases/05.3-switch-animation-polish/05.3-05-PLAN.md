3. Add the `switchAnimation` private method:

```typescript
/**
 * Play 3-frame switch-in animation: pokeball -> filled square -> checkbox -> real sprite.
 * No artificial delays — each await glass.updateImage() IS the frame timing.
 * After the real sprite appears, slide it 2-3px left-to-right.
 */
private async switchAnimation(
  side: Side,
  nextState: BattleState,
  glass: GlassAdapter,
): Promise<void> {
  const containerID = side === 'opponent'
    ? CONTAINER_IDS.oppSprite
    : CONTAINER_IDS.playerSprite;
  const containerName = side === 'opponent'
    ? CONTAINER_NAMES.oppSprite
    : CONTAINER_NAMES.playerSprite;

  const updateImg = async (data: Uint8Array): Promise<void> => {
    await glass.updateImage(containerID, containerName, data);
  };

  if (this.cancelled) return;

  // Frame 1: Pokeball
  await updateImg(getPokeballFrame());
  if (this.cancelled) return;

  // Frame 2: Filled square (energy burst)
  await updateImg(getFilledSquareFrame());
  if (this.cancelled) return;

  // Frame 3: Checkbox (materialization)
  await updateImg(getCheckboxFrame());
  if (this.cancelled) return;

  // Frame 4: Real pokemon sprite
  const newSprite = getSpriteForSide(nextState, side);
  if (newSprite) {
    await updateImg(newSprite);
  }

  // Sprite slide effect: The G2 SDK doesn't support pixel offsets on existing
  // containers, so the "slide" is simulated by a brief blank->sprite sequence
  // that gives the visual impression of motion. This is the best approximation
  // on the G2 hardware without container repositioning.
  // The sprite is already displayed — no additional slide needed if hardware
  // doesn't support sub-container positioning.
}
```

4. Add the `hpHeal` private method (reverse of hpDrain):

```typescript
/**
 * Reverse HP drain — HP bar fills UP progressively.
 * Used for HEAL entries (Leftovers, Wish, Recover, etc.)
 */
private async hpHeal(
  side: Side,
  entry: TurnLogEntry,
  prevState: BattleState,
  glass: GlassAdapter,
  setHpStatus: (status: string) => void,
  triggerUpdate: () => Promise<void>,
): Promise<void> {
  const targetHp = entry.detail.targetHp as number | undefined;
  const targetMaxHp = entry.detail.targetMaxHp as number | undefined;
  if (typeof targetHp !== 'number' || typeof targetMaxHp !== 'number') return;
  if (targetMaxHp <= 0) return;

  const mon = getActiveMon(prevState, side);
  const startHp = mon.currentHp;
  // Calculate healed HP from log values (percentage-based for opponent)
  const healedPct = targetHp / targetMaxHp;
  const endHp = Math.min(mon.stats.hp, Math.round(healedPct * mon.stats.hp));
  const healAmount = endHp - startHp;

  if (healAmount <= 0) return; // No actual heal

  const steps = Math.min(healAmount, ANIM.MAX_HP_STEPS);
  if (steps <= 0) return;

  for (let i = 0; i <= steps; i++) {
    if (this.cancelled) break;

    const currentHp = Math.round(startHp + (healAmount * i / steps));
    const tempMon: ActivePokemon = { ...mon, currentHp, fainted: false };

    setHpStatus(formatHpBar(tempMon));
    await triggerUpdate();
    await this.delay(ANIM.HP_FRAME_MS);
  }
}
```

```typescript
private async resolveSwitch(toIndex: number): Promise<void> {
  const prevState = this.state.battleState!;
  const log: TurnLogEntry[] = [];
  const battle = SimBattleAdapter.getBattle()!;

  let bs: BattleState;

  // Check if opponent's active is also fainted (double faint)
  const opponentSide = prevState.opponentSide;
  const opponentActive = opponentSide.team[opponentSide.activeIndex]!;
  if (opponentActive.fainted) {
    // AI picks its switch target
    const aiSwitchIndex = SimAIOpponent.pickSwitch(battle);
    // Submit both switches together via the aiSwitchIndex parameter
    const result = SimBattleAdapter.submitSwitch(prevState, 'player', toIndex, aiSwitchIndex);
    log.push(...result.log);
    bs = result.nextState;
  } else {
    // Only player needs to switch
    const result = SimBattleAdapter.submitSwitch(prevState, 'player', toIndex);
    log.push(...result.log);
    bs = result.nextState;
  }

  this.lastTurnLog = log;
  this.state = {
    ...this.state,
    battleState: bs,
    cursorIndex: 0,
  };

  // Step 1: Show HUD with prevState (sprites before switch)
  const prevHud = renderBattleHud(prevState);
  await this.glass.showBattleHud(
    prevHud.oppText,
    prevHud.playerText,
    prevHud.status,
    prevHud.oppSprite,
    prevHud.playerSprite,
  );

  // Step 2: Play switch animation (3-frame pokeball + text)
  await this.animationEngine.playTurn(log, prevState, bs, this.glass);

  // Step 3: Show HUD with final state (new sprites, correct HP)
  // Preload sprites for the newly switched-in Pokemon
  await this.preloadSprites(bs);

  const finalHud = renderBattleHud(bs);
  await this.glass.showBattleHud(
    finalHud.oppText,
    finalHud.playerText,
    finalHud.status,
    finalHud.oppSprite,
    finalHud.playerSprite,
  );

  // Step 4: Brief pause before transition
  await this.delay(ANIM.POST_ANIM_DELAY_MS);

  // Step 5: Transition based on resulting phase
  await this.transitionAfterTurn(bs);
}
```

```typescript
/** Preload BMP sprites for both active Pokemon in the given state. */
private async preloadSprites(state: BattleState): Promise<void> {
  const getDexNum = (species: string): number => {
    const slug = species.toLowerCase().replace(/[^a-z0-9]/g, '');
    return DataLoader.getPokemon(slug)?.num ?? 0;
  };

  const playerMon = state.playerSide.team[state.playerSide.activeIndex]!;
  const oppMon = state.opponentSide.team[state.opponentSide.activeIndex]!;
  const playerDex = getDexNum(playerMon.species);
  const oppDex = getDexNum(oppMon.species);

  const promises: Promise<unknown>[] = [];
  if (playerDex > 0 && !SpriteService.getBackBitmapCached(playerDex)) {
    promises.push(SpriteService.preloadBack(playerDex));
  }
  if (oppDex > 0 && !SpriteService.getBitmapCached(oppDex)) {
    promises.push(SpriteService.preload(oppDex));
  }
  await Promise.all(promises);
}
```
