```typescript
/** Format name with tera type, status, and volatile conditions in brackets.
 *  Examples: "You: Pikachu (Fire | BRN, CNF)" / "You: Pikachu (BRN)" / "You: Pikachu" */
function formatName(label: string, mon: ActivePokemon): string {
  const parts: string[] = [];

  // 1. Tera type first
  if (mon.teraActive) parts.push(mon.teraType);

  // 2. Non-volatile status
  if (mon.status) parts.push(mon.status.toUpperCase());

  // 3. Volatile statuses
  if (mon.volatiles && mon.volatiles.length > 0) {
    parts.push(...mon.volatiles);
  }

  if (parts.length === 0) return `${label}: ${mon.name}`;

  // Tera type separated by " | " from statuses for readability
  let partsStr: string;
  if (mon.teraActive && parts.length > 1) {
    // "Fire | BRN, CNF"
    partsStr = parts[0] + ' | ' + parts.slice(1).join(', ');
  } else {
    partsStr = parts.join(', ');
  }

  const base = `${label}: ${mon.name} (`;
  const maxLen = 64; // DISPLAY_BUDGET.LINE_WIDTH
  const suffix = ')';

  // Truncate if too long
  if (base.length + partsStr.length + suffix.length > maxLen) {
    const available = maxLen - base.length - suffix.length - 3; // 3 for "..."
    if (available > 0) {
      partsStr = partsStr.slice(0, available) + '...';
    } else {
      // Name itself is too long, just show name
      return `${label}: ${mon.name}`;
    }
  }

  return `${base}${partsStr}${suffix}`;
}
```
