```typescript
import type { BattleState } from '../types/battleState.js';

export type ScreenName =
  | 'MAIN_MENU'
  | 'BATTLE_HUD'
  | 'ACTION_MENU'
  | 'ATTACK_SELECT'
  | 'SWITCH_SELECT'
  | 'TERA_CONFIRM'
  | 'TURN_RESULT'
  | 'END_SCREEN';

export interface ScreenState {
  screen: ScreenName;
  cursorIndex: number;
  battleState: BattleState | null;
}

// Return types for renderers
export interface TextScreen {
  text: string;
  status: { left: string; right: string };
}

export interface ListScreen {
  items: string[];
  status: string;
}

// Extended list screen that maps display index to team index
export interface SwitchListScreen extends ListScreen {
  indexMap: number[];  // indexMap[displayIndex - 1] = team index (offset -1 because index 0 is "< Back")
}
```

**mainMenuRenderer.ts** (UI-01) -- List screen:

```typescript
import type { ListScreen } from '../ScreenState.js';

export function renderMainMenu(): ListScreen {
```

**battleHudRenderer.ts** (UI-02) -- Text screen:
- Takes `BattleState`, returns `TextScreen`
- Shows opponent name + HP% on top, player name + HP% on bottom
- Truncate names to 12 chars with `.slice(0, 12).padEnd(12)`
- HP% uses `Math.max(1, Math.round((currentHp / stats.hp) * 100))` -- NEVER show 0% for non-fainted
- If teraActive, append ` [TeraType]` after HP%
- Status bar: left = `Turn ${turnCount}`, right = `Tap for menu`
- Status conditions: if pokemon has a non-empty status, show it as abbreviation after HP% (e.g., `47% BRN`)
- Enforce: total text must be < 900 chars (slice if needed, though battle HUD is always short)

**actionMenuRenderer.ts** (UI-03) -- List screen:
- Takes `{ teraUsed: boolean }` as parameter
- Returns `{ items: ['< Back', 'Attack', 'Switch', teraLabel], status: 'Choose action' }`
- If `teraUsed` is true, set teraLabel to `'Tera (used)'`; otherwise `'Terastallize'`
- ScreenController will handle disabling, but renderer visually indicates tera status
```typescript
export function renderActionMenu(opts: { teraUsed: boolean }): ListScreen {
  return {
    items: [
      '< Back',
      'Attack',
      'Switch',
      opts.teraUsed ? 'Tera (used)' : 'Terastallize',
    ],
    status: 'Choose action',
  };
}
```

**attackSelectRenderer.ts** (UI-04) -- List screen:
- Takes `BattleState`, returns `ListScreen`
- First item is `'< Back'`
- For each of the active Pokemon's 4 moves: `DataLoader.getMove(slug)` to get name, type, pp
- Format: `"{MoveName} ({Type}) {remaining}/{max}"` -- e.g., `"Thunderbolt (Electric) 12/16"`
- Enforce each item fits within 64 chars (truncate move name if needed)
- If a move has 0 PP remaining, append ` [--]` to indicate unusable
- Status: `'Choose move'`

**switchSelectRenderer.ts** (UI-05) -- SwitchListScreen:
- Takes `(state: BattleState, forced: boolean)`, returns `SwitchListScreen`
- First item is `'< Back'` (even on forced switch -- ScreenController will handle ignoring back on forced)
- Iterate through player's team, SKIP `i === side.activeIndex` (current active) AND `mon.fainted === true`
- For each eligible Pokemon: `"{name} {hpPct}%"` with name truncated to 12 chars
- Build `indexMap: number[]` where `indexMap[displayIndex - 1]` maps back to the original team index
- Status: `forced ? 'Switch in!' : 'Switch to'`

**teraConfirmRenderer.ts** (UI-06) -- List screen:
- Takes `BattleState`, returns `ListScreen`
- Shows Tera type of active player Pokemon
- Items: `['< Back', 'Confirm Terastallize']`
- Status: `'Tera: {teraType}'` where teraType comes from active Pokemon's `.teraType`

**turnResultRenderer.ts** (UI-07) -- Text screen:
- Takes `TurnLogEntry[]`, returns `TextScreen`
- Format each log entry with a fixed template based on `entry.kind`:
  - DAMAGE: `"{Actor}'s {pokemon} used {move}! {damage} dmg"` -- where actor is derived and pokemon/move from detail
  - MISS: `"{Actor}'s attack missed!"`
  - FAINT: `"{pokemon} fainted!"`
  - STATUS_INFLICT: `"{pokemon} was {status}!"`
  - STATUS_TICK: `"{pokemon} took {status} damage!"`
  - SWITCH: `"Go! {pokemon}!"`
  - TERA: `"{Actor} Terastallized! [{teraType}]"`
  - WIN: `"You win!"`
  - LOSS: `"You lost!"`
  - PP_OUT: `"No PP left for {move}!"`
  - PARALYSIS_SKIP: `"{pokemon} is paralyzed!"`
  - FROZEN: `"{pokemon} is frozen solid!"`
  - ASLEEP: `"{pokemon} is fast asleep!"`
- Show only last `DISPLAY_BUDGET.MAX_LOG_ENTRIES` entries (6)
- Join with `\n`, enforce total < 900 chars
- Status: `{ left: 'Turn result', right: 'Tap to continue' }`

For the actor labels in turn result, use simple mapping: `actor === 'player' ? 'Your' : "Foe's"` prefix. For pokemon names, pull from `detail.pokemon` when available.

**endScreenRenderer.ts** (UI-08) -- List screen:
- Takes `(winner: 'player' | 'opponent')`, returns `ListScreen`
- Items: `['New Battle', 'Main Menu']` (no back -- this is a terminal screen)
- Status: `winner === 'player' ? 'Victory!' : 'Defeat...'`

**All renderers must:**
- Import types from `../ScreenState.js` (TextScreen, ListScreen, SwitchListScreen)
- Be pure functions (no side effects, no glass/engine calls)
- Use `.js` extension in import paths (ESM convention in this project)
- Use DISPLAY_BUDGET constants from `../../config/constants.js` where applicable
